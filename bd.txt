    ux_user_types_type_name
ON user_types(type_name);


INSERT INTO user_types (type_name, description)
SELECT v.type_name, v.description
FROM (
    VALUES
        ('admin', 'Administrador do sistema'),
        ('cliente', 'Cliente que busca profissionais e gerencia projetos'),

-- ============================================================
-- USUÁRIOS
-- ============================================================

CREATE TABLE IF NOT EXISTS users (
    id UUID PRIMARY KEY
        REFERENCES auth.users(id)
        ON DELETE CASCADE,

    email VARCHAR(255) NOT NULL,
    full_name VARCHAR(255) NOT NULL,
    phone VARCHAR(20),
    avatar_url TEXT,

    user_type_id INTEGER
        REFERENCES user_types(id),

    is_verified BOOLEAN DEFAULT FALSE,
    is_active BOOLEAN DEFAULT TRUE,

    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE UNIQUE INDEX IF NOT EXISTS
    ux_users_email
ON users(email);

CREATE INDEX IF NOT EXISTS
    idx_users_type
ON users(user_type_id);

    idx_users_active

-- ============================================================
-- PROPOSTAS
-- ============================================================

CREATE TABLE IF NOT EXISTS proposals (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),

    project_id UUID NOT NULL
        REFERENCES projects(id)
        ON DELETE CASCADE,

    sender_type VARCHAR(20) NOT NULL
        CHECK (
            sender_type IN (
                'professional',
                'contractor'
            )
        ),

    professional_id UUID
        REFERENCES professional_profiles(id),

    contractor_id UUID
        REFERENCES contractor_profiles(id),

    title VARCHAR(255) NOT NULL,
    description TEXT NOT NULL,

    proposed_value DECIMAL(15,2) NOT NULL,
    proposed_duration_days INTEGER,

    scope_description TEXT,

    materials_included BOOLEAN DEFAULT FALSE,
    materials_value DECIMAL(15,2),
    labor_value DECIMAL(15,2),

    payment_terms TEXT,

    warranty_months INTEGER DEFAULT 0,

    start_date_availability DATE,

    status_id INTEGER
        REFERENCES proposal_status(id)
        DEFAULT 1,

    sent_at TIMESTAMPTZ DEFAULT NOW(),
    expires_at TIMESTAMPTZ,

    responded_at TIMESTAMPTZ,
    response_message TEXT,

    revision_count INTEGER DEFAULT 0,

    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW(),

    CONSTRAINT chk_proposal_sender
    CHECK (
        (
            sender_type = 'professional'
            AND professional_id IS NOT NULL
            AND contractor_id IS NULL
        )
        OR
        (
            sender_type = 'contractor'
            AND contractor_id IS NOT NULL
            AND professional_id IS NULL
        )
    )
);

CREATE INDEX IF NOT EXISTS
    idx_proposals_project
ON proposals(project_id);

CREATE INDEX IF NOT EXISTS
    idx_proposals_status
ON proposals(status_id);

CREATE INDEX IF NOT EXISTS
    idx_proposals_professional
ON proposals(professional_id);

CREATE INDEX IF NOT EXISTS
    idx_proposals_contractor
ON proposals(contractor_id);

-- PROGRESSO
-- ============================================================

CREATE TABLE IF NOT EXISTS project_progress (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),

    project_id UUID NOT NULL
        REFERENCES projects(id)
        ON DELETE CASCADE,

    stage_id INTEGER
        REFERENCES project_stages(id),

    status VARCHAR(20) DEFAULT 'pendente'
        CHECK (
            status IN (
                'pendente',
                'em_andamento',
                'concluido',
                'atrasado'
            )
        ),

    progress_percentage INTEGER DEFAULT 0
        CHECK (
            progress_percentage >= 0
            AND progress_percentage <= 100
        ),

    planned_start_date DATE,
    planned_end_date DATE,

    actual_start_date DATE,
    actual_end_date DATE,

    responsible_professional_id UUID
        REFERENCES professional_profiles(id),

    responsible_contractor_id UUID
        REFERENCES contractor_profiles(id),

    description TEXT,
    notes TEXT,

    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW(),

    updated_by UUID
        REFERENCES users(id)
);

CREATE INDEX IF NOT EXISTS
    idx_project_progress_project
ON project_progress(project_id);

CREATE INDEX IF NOT EXISTS

-- ============================================================
-- RATING PROFISSIONAL
-- ============================================================

CREATE OR REPLACE FUNCTION update_professional_rating()
RETURNS TRIGGER
LANGUAGE plpgsql
AS $$
DECLARE
    target_professional UUID;
BEGIN

    IF TG_OP = 'DELETE' THEN

        target_professional := OLD.professional_id;

        UPDATE professional_profiles
        SET
            average_rating = COALESCE(
                (
                    SELECT ROUND(AVG(rating)::numeric, 1)
                    FROM professional_reviews
                    WHERE professional_id = target_professional
                      AND is_approved = TRUE
                ),
                0
            ),
            total_reviews = (
                SELECT COUNT(*)
                FROM professional_reviews
                WHERE professional_id = target_professional
                  AND is_approved = TRUE
            )
        WHERE id = target_professional;

        RETURN OLD;

    ELSIF TG_OP = 'UPDATE' THEN

        IF OLD.professional_id IS DISTINCT FROM NEW.professional_id THEN

            UPDATE professional_profiles
            SET
                average_rating = COALESCE(
                    (
                        SELECT ROUND(AVG(rating)::numeric, 1)
                        FROM professional_reviews
                        WHERE professional_id = OLD.professional_id
                          AND is_approved = TRUE
                    ),
                    0
                ),
                total_reviews = (
                    SELECT COUNT(*)
                    FROM professional_reviews
                    WHERE professional_id = OLD.professional_id
                      AND is_approved = TRUE
                )
            WHERE id = OLD.professional_id;

        END IF;

        target_professional := NEW.professional_id;

    ELSE

        target_professional := NEW.professional_id;

    END IF;


    UPDATE professional_profiles
    SET
        average_rating = COALESCE(
            (
                SELECT ROUND(AVG(rating)::numeric, 1)
                FROM professional_reviews
                WHERE professional_id = target_professional
                  AND is_approved = TRUE
            ),
            0
        ),
        total_reviews = (
            SELECT COUNT(*)
            FROM professional_reviews
            WHERE professional_id = target_professional
              AND is_approved = TRUE
        )
    WHERE id = target_professional;

    RETURN NEW;

END;
$$;


DROP TRIGGER IF EXISTS update_professional_rating_after_change
ON professional_reviews;

CREATE TRIGGER update_professional_rating_after_change
AFTER INSERT OR UPDATE OR DELETE
ON professional_reviews
FOR EACH ROW
EXECUTE FUNCTION update_professional_rating();

ON contractor_reviews;

ON contractor_reviews
FOR INSERT
WITH CHECK (
    auth.uid() = reviewer_id
);


-- ============================================================
-- VIEWS
-- ============================================================

CREATE OR REPLACE VIEW professional_profiles_complete AS
SELECT
    pp.*,

    u.full_name,
    u.email,
    u.phone,
    u.avatar_url,
    u.is_verified,

    COALESCE(
        ARRAY_AGG(DISTINCT s.specialty_name)
        FILTER (
            WHERE s.specialty_name IS NOT NULL
        ),
        ARRAY[]::VARCHAR[]
    ) AS specialties

FROM professional_profiles pp

JOIN users u
    ON pp.user_id = u.id

LEFT JOIN professional_specialties ps
    ON pp.id = ps.professional_id

LEFT JOIN specialties s
    ON ps.specialty_id = s.id

GROUP BY
    pp.id,
    u.id;


-- ============================================================
-- VIEW DE PROJETOS
-- ============================================================

CREATE OR REPLACE VIEW projects_complete AS
SELECT
    p.*,

    u.full_name AS client_name,
    u.avatar_url AS client_avatar,

    pc.category_name,

    ps.status_name,
    ps.color_code AS status_color

FROM projects p

JOIN users u
    ON p.client_id = u.id

LEFT JOIN project_categories pc
    ON p.category_id = pc.id

LEFT JOIN project_status ps
    ON p.status_id = ps.id;


-- ============================================================
-- VIEW DASHBOARD
-- ============================================================

CREATE OR REPLACE VIEW dashboard_stats AS
SELECT

    (
        SELECT COUNT(*)
        FROM users
        WHERE is_active = TRUE
    ) AS total_users,

    (
        SELECT COUNT(*)
        FROM professional_profiles
        WHERE is_available = TRUE
    ) AS active_professionals,

    (
        SELECT COUNT(*)
        FROM contractor_profiles
    ) AS total_contractors,

    (
        SELECT COUNT(*)
        FROM projects
        WHERE status_id = (
            SELECT id
            FROM project_status
            WHERE status_name = 'em_andamento'
            LIMIT 1
        )
    ) AS projects_in_progress,

    (
        SELECT COUNT(*)
        FROM projects
        WHERE status_id = (
            SELECT id
            FROM project_status
            WHERE status_name = 'concluido'
            LIMIT 1
        )
    ) AS projects_completed,

    (
        SELECT COUNT(*)
        FROM proposals
        WHERE status_id = (
            SELECT id
            FROM proposal_status
            WHERE status_name = 'aceita'
            LIMIT 1
        )
    ) AS proposals_accepted,

    (
        SELECT COALESCE(
            AVG(rating),
            0
        )
        FROM professional_reviews
        WHERE is_approved = TRUE
    ) AS avg_professional_rating;
